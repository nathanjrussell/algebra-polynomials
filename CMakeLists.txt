cmake_minimum_required(VERSION 3.24)
project(algebra_polynomials VERSION 0.2.0 LANGUAGES CXX)
option(ALGEBRA_POLYNOMIALS_BUILD_EXAMPLES "Build examples" ${PROJECT_IS_TOP_LEVEL})
option(ALGEBRA_POLYNOMIALS_BUILD_TESTS "Build tests" ${PROJECT_IS_TOP_LEVEL})
include(cmake/Dependencies.cmake)
add_subdirectory(src)
if(ALGEBRA_POLYNOMIALS_BUILD_EXAMPLES OR ALGEBRA_POLYNOMIALS_BUILD_TESTS)
    include(cmake/DevelopmentDependencies.cmake)
endif()
if(ALGEBRA_POLYNOMIALS_BUILD_EXAMPLES)
    add_subdirectory(example)
endif()
if(ALGEBRA_POLYNOMIALS_BUILD_TESTS)
    enable_testing()
    include(cmake/GoogleTest.cmake)
    add_subdirectory(test)
endif()
